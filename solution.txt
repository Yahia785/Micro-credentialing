def solution():
    # Read the array
    nums = list(map(int, input().split()))
    
    # Read the target
    target = int(input())
    
    # Your code here
    # Find two indices i and j where nums[i] + nums[j] = target
    # Store the result as: result = f"{i} {j}" where i < j
    
    seen = {}  # value -> index
    i = j = -1
    for idx, val in enumerate(nums):
        complement = target - val
        if complement in seen:
            i, j = seen[complement], idx
            break
        seen[val] = idx
    
    if i > j:
        i, j = j, i

    result = f"{i} {j}"
    
    # Print the result
    print(result)

if __name__ == "__main__":
    solution()
